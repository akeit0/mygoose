cmake_minimum_required(VERSION 3.20)
project(mygoose_native LANGUAGES C)
if(WIN32)
    add_library(mygoose_native_impl STATIC native/windows/platform.c)
    target_link_libraries(mygoose_native_impl PUBLIC user32 gdi32 imm32 d2d1 dwrite)
elseif(APPLE)
    enable_language(OBJC)
    add_library(mygoose_native_impl STATIC native/macos/platform.m)
    target_link_libraries(mygoose_native_impl PUBLIC "-framework Cocoa" "-framework CoreText" "-framework CoreGraphics" "-framework QuartzCore")
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(DESKTOP REQUIRED IMPORTED_TARGET x11 pangocairo)
    add_library(mygoose_native_impl STATIC native/linux/platform.c)
    target_link_libraries(mygoose_native_impl PUBLIC PkgConfig::DESKTOP m)
else()
    message(FATAL_ERROR "MyGoose has backends for Windows, Linux and macOS only")
endif()
set_target_properties(mygoose_native_impl PROPERTIES C_STANDARD 11 POSITION_INDEPENDENT_CODE ON)
target_include_directories(mygoose_native_impl PUBLIC native)
add_library(mygoose_native INTERFACE)
if(MINGW)
    # Goose 448efe7 copies CMake's pkg\libfoo.a into a GCC response file
    # without escaping the backslash. Explicit -L/-l avoids that path spelling.
    # Reproducer/status: docs/goose-issues.md (GOOSE-001).
    add_dependencies(mygoose_native mygoose_native_impl)
    target_link_libraries(mygoose_native INTERFACE "-L${CMAKE_CURRENT_BINARY_DIR}" -lmygoose_native_impl user32 gdi32 imm32 d2d1 dwrite)
else()
    target_link_libraries(mygoose_native INTERFACE mygoose_native_impl)
endif()
