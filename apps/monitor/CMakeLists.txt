cmake_minimum_required(VERSION 3.20)
project(mygoose_monitor LANGUAGES C)
if(WIN32)
    add_library(mygoose_monitor_native_impl STATIC native/windows.c native/icons_windows.c)
    target_link_libraries(mygoose_monitor_native_impl PUBLIC psapi iphlpapi pdh advapi32 shell32 ole32)
elseif(APPLE)
    enable_language(OBJC)
    add_library(mygoose_monitor_native_impl STATIC native/macos.c native/icons_macos.m)
    target_link_libraries(mygoose_monitor_native_impl PUBLIC "-framework IOKit" "-framework Cocoa")
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_library(mygoose_monitor_native_impl STATIC native/linux.c native/icons_other.c)
    target_link_libraries(mygoose_monitor_native_impl PUBLIC m)
else()
    message(FATAL_ERROR "Monitor supports Windows, Linux and macOS")
endif()
set_target_properties(mygoose_monitor_native_impl PROPERTIES C_STANDARD 11 POSITION_INDEPENDENT_CODE ON)
add_library(mygoose_monitor_native INTERFACE)
if(MINGW)
    # As in the core package, avoid Goose's unescaped CMake backslash paths.
    # See docs/goose-issues.md (GOOSE-001) in the repository root.
    add_dependencies(mygoose_monitor_native mygoose_monitor_native_impl)
    target_link_libraries(mygoose_monitor_native INTERFACE "-L\"${CMAKE_CURRENT_BINARY_DIR}\"" -lmygoose_monitor_native_impl psapi iphlpapi pdh advapi32 shell32 ole32)
else()
    target_link_libraries(mygoose_monitor_native INTERFACE mygoose_monitor_native_impl)
endif()
